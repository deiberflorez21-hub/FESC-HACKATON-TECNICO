# Diagnóstico:
# 1. Delimitador punto y coma (;) en vez de coma (,).
# 2. Documentos con padding de ceros a la izquierda (longitud 12).
# 3. Situación VIGENTE (equivale a 'A') y RETIRADO (equivale a 'I').
# Cruce contra AFI_MAE.csv:
$ python3 -c "
import csv
situacion_map = {'VIGENTE': 'A', 'RETIRADO': 'I'}
afi = {}
for r in csv.DictReader(open('data/AFI_MAE.csv')):
    d = r['NRO_DOC'].strip()
    if d not in afi:
        afi[d] = r['EST_AFI'].strip()

mismatches = 0
for r in csv.DictReader(open('data/extracto_caja_oriente.csv'), delimiter=';'):
    doc = r['id_afiliado'].strip().lstrip('0')
    sit = r['situacion'].strip()
    if doc not in afi or afi[doc] != situacion_map.get(sit):
        mismatches += 1

print('Discrepancias de cruce:', mismatches)
"
Discrepancias de cruce: 25
