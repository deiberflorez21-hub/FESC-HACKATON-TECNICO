from datetime import datetime
import csv

def parse_date(s):
    s = s.strip()
    for fmt in ('%Y-%m-%d', '%d/%m/%Y', '%Y%m%d'):
        try:
            return datetime.strptime(s, fmt).strftime('%Y-%m-%d')
        except ValueError:
            pass
    return None

seen = set()
invalid = []
for r in csv.DictReader(open('data/AFI_MAE.csv')):
    d = r['NRO_DOC'].strip()
    if d not in seen:
        seen.add(d)
        if not parse_date(r['FEC_AFI']):
            invalid.append((d, r['FEC_AFI']))

print('Total fechas inválidas irreconciliables:', len(invalid))
for doc, fec in invalid[:5]:
    print(f'  Doc: {doc}, Fecha: {fec}')
