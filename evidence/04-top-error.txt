# Diagnóstico: Extracción de códigos de error CIE-#### y conteo por frecuencia
$ grep -hoE 'CIE-[0-9]{4}' logs/*.log | sort | uniq -c | sort -nr
   1424 CIE-4012
    700 CIE-5003
    460 CIE-2210
    298 CIE-3301
    158 CIE-1107
     75 CIE-6020

# Solución: El error más frecuente es CIE-4012 con 1424 apariciones.
# Generación automatizada en output/top_error.txt:
$ grep -hoE 'CIE-[0-9]{4}' logs/*.log | sort | uniq -c | sort -nr | head -n 1 | awk '{print "codigo_error=" $2 "\nocurrencias=" $1}' > output/top_error.txt

$ cat output/top_error.txt
codigo_error=CIE-4012
ocurrencias=1424
